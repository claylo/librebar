- [ ] **Step 4: Collect response parts and trailer frames**

Change body collection to return both data and trailers:

```rust
async fn read_body(
    mut body: ResponseBody,
    maximum: usize,
) -> std::result::Result<(Vec<u8>, Option<HeaderMap>), ReadBodyError> {
    let mut output = Vec::new();
    let mut trailers: Option<HeaderMap> = None;
    while let Some(frame) = body.frame().await {
        let frame = frame.map_err(ReadBodyError::Body)?;
        match frame.into_data() {
            Ok(data) => append_bounded(&mut output, &data, maximum)?,
            Err(frame) => {
                if let Ok(next) = frame.into_trailers() {
                    let stored = trailers.get_or_insert_with(HeaderMap::new);
                    for (name, value) in next {
                        if let Some(name) = name {
                            stored.append(name, value);
                        }
                    }
                }
            }
        }
    }
    Ok((output, trailers))
}

fn append_bounded(
    output: &mut Vec<u8>,
    data: &[u8],
    maximum: usize,
) -> std::result::Result<(), ReadBodyError> {
    let new_size = output.len().checked_add(data.len()).ok_or(ReadBodyError::TooLarge)?;
    if maximum > 0 && new_size > maximum {
        return Err(ReadBodyError::TooLarge);
    }
    output.extend_from_slice(data);
    Ok(())
}
```

In `send`, split the Hyper response before body collection and construct `ResponseMetadata` from `parts.status`, `parts.version`, `parts.headers`, and collected trailers. Keep retry decisions based on the original status.

```rust
#[derive(Clone, Debug, Eq, PartialEq)]
pub struct Validator {
    etag: Option<HeaderValue>,
    last_modified: Option<HeaderValue>,
}

impl Validator {
    pub fn etag(value: HeaderValue) -> Self { Self { etag: Some(value), last_modified: None } }
    pub fn last_modified(value: HeaderValue) -> Self { Self { etag: None, last_modified: Some(value) } }
    pub fn from_headers(headers: &HeaderMap) -> Option<Self> {
        let etag = headers.get(hyper::header::ETAG).cloned();
        let last_modified = headers.get(hyper::header::LAST_MODIFIED).cloned();
        (etag.is_some() || last_modified.is_some()).then_some(Self { etag, last_modified })
    }
    pub const fn etag(&self) -> Option<&HeaderValue> { self.etag.as_ref() }
    pub const fn last_modified(&self) -> Option<&HeaderValue> { self.last_modified.as_ref() }
}

pub enum ConditionalResponse {
    Modified(Response),
    NotModified(ResponseMetadata),
    Indeterminate(Response),
}

pub enum ModificationCheck {
    Modified(ResponseMetadata),
    NotModified(ResponseMetadata),
    Indeterminate(ResponseMetadata),
}
```

```rust
fn conditional_request(
    method: Method,
    url: &str,
    validator: &Validator,
) -> Result<Request<Bytes>> {
    let mut builder = Request::builder().method(method).uri(url);
    if let Some(etag) = validator.etag() {
        builder = builder.header(hyper::header::IF_NONE_MATCH, etag);
    } else if let Some(last_modified) = validator.last_modified() {
        builder = builder.header(hyper::header::IF_MODIFIED_SINCE, last_modified);
    }
    builder.body(Bytes::new()).map_err(HttpError::RequestBuild).map_err(Into::into)
}
```
